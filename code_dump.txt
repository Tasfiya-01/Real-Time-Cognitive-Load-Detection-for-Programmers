
########## src\train_xgb.py ##########

"""Clean XGBoost training: no scaler, constant columns dropped, two evaluation protocols.
Run from the project root:  python src\\train_xgb.py
Uses the same load_xy() / FEATURE_FILE as loso_eval.py, so all thesis numbers share one source.
"""
import numpy as np
import joblib
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.model_selection import StratifiedShuffleSplit, GroupShuffleSplit
from sklearn.metrics import accuracy_score, classification_report, confusion_matrix
from xgboost import XGBClassifier
from common import *

MODELS = ROOT / "models"
MODELS.mkdir(exist_ok=True)

print("Loading features...")
X, y, groups = load_xy()
classes = [str(c) for c in np.unique(np.load(DATA / "y_labels.npy"))]
print("Classes in index order (0,1,2,...):", classes)

# Drop (near-)constant columns (the 56 Mean/Std/Var/RMS columns). Label-free.
keep = np.flatnonzero(X.std(axis=0) > 1e-6)
X = X[:, keep]
print(f"Kept {len(keep)} features, X shape: {X.shape}")
np.save(MODELS / "keep_idx.npy", keep)   # the SHAP script needs this to map names back


def make_model():
    return XGBClassifier(n_estimators=N_EST, max_depth=6, learning_rate=0.1,
                         eval_metric="mlogloss", random_state=SEED, n_jobs=-1)


def run(title, tr, te):
    m = make_model()
    m.fit(X[tr], y[tr])
    p = m.predict(X[te])
    acc = accuracy_score(y[te], p)
    print("\n" + "=" * 60)
    print(f"{title}: {acc*100:.2f}%  (train {len(tr)}, test {len(te)})")
    print("=" * 60)
    print(classification_report(y[te], p, target_names=classes, zero_division=0))
    return m, p, acc


# 1) Subject-mixed stratified 80/20: the old headline protocol (segments of the same
#    subject appear in train AND test, so it is leaky).
tr, te = next(StratifiedShuffleSplit(1, test_size=0.2, random_state=SEED).split(X, y))
m_mixed, p_mixed, acc_mixed = run("Subject-MIXED 80/20 (leaky)", tr, te)

# 2) Subject-held-out 80/20: whole subjects are test-only. Single split of ~9 subjects,
#    so it is noisy; the LOSO numbers from loso_eval.py are the reliable ones.
tr2, te2 = next(GroupShuffleSplit(1, test_size=0.2, random_state=SEED).split(X, y, groups))
m_subj, p_subj, acc_subj = run("Subject-HELD-OUT 80/20", tr2, te2)

print("\nSaving model...")
joblib.dump(m_mixed, MODELS / "xgb_model_clean.pkl")
print("Saved models/xgb_model_clean.pkl and models/keep_idx.npy")

fig, axes = plt.subplots(1, 2, figsize=(12, 5))
for ax, (t, yt, pt, a) in zip(axes, [("Subject-mixed (leaky)", y[te], p_mixed, acc_mixed),
                                     ("Subject-held-out", y[te2], p_subj, acc_subj)]):
    sns.heatmap(confusion_matrix(yt, pt), annot=True, fmt="d", cmap="Blues", ax=ax,
                xticklabels=classes, yticklabels=classes)
    ax.set_title(f"{t}\nAccuracy: {a*100:.1f}%")
    ax.set_xlabel("Predicted")
    ax.set_ylabel("Actual")
plt.tight_layout()
plt.savefig(OUT / "xgb_results_clean.png", dpi=150)
print("Plot saved: results/xgb_results_clean.png")

########## src\old_features_v3.py ##########

import numpy as np
from scipy.signal import welch
from scipy.stats import skew, kurtosis
import antropy as ant

FS = 128

def extract_features_v3(segment):
    features = []

    for ch in range(segment.shape[1]):
        ch_data = np.ascontiguousarray(segment[:, ch], dtype=np.float64)
        freqs, psd = welch(ch_data, fs=FS, nperseg=256)

        def bp(low, high):
            idx = np.where((freqs >= low) & (freqs < high))[0]
            return np.mean(psd[idx])

        delta = bp(0.5, 4)
        theta = bp(4, 8)
        alpha = bp(8, 13)
        beta  = bp(13, 30)
        gamma = bp(30, 50)

        theta_alpha = theta / (alpha + 1e-8)
        beta_alpha  = beta  / (alpha + 1e-8)
        theta_beta  = theta / (beta  + 1e-8)

        mean = np.mean(ch_data)
        std  = np.std(ch_data)
        var  = np.var(ch_data)
        sk   = skew(ch_data)
        kurt = kurtosis(ch_data)
        rms  = np.sqrt(np.mean(ch_data**2))

        try:
            samp_en  = ant.sample_entropy(ch_data)
            perm_en  = ant.perm_entropy(ch_data, normalize=True)
            spec_en  = ant.spectral_entropy(ch_data, sf=FS, normalize=True)
            svd_en   = ant.svd_entropy(ch_data, normalize=True)
        except Exception as e:
            raise RuntimeError(f"entropy failed: {e}")

        features.extend([
            delta, theta, alpha, beta, gamma,
            theta_alpha, beta_alpha, theta_beta,
            mean, std, var, sk, kurt, rms,
            samp_en, perm_en, spec_en, svd_en
        ])

    return np.array(features)


if __name__ == '__main__':
    import scipy.io as sio
    from scipy import signal as sig

    print("Loading data...")
    dataset = sio.loadmat('data/dataset.mat')
    labels  = sio.loadmat('data/class_012.mat')

    eeg_data     = dataset['dataset']
    class_labels = labels['class_012'].flatten()

    data_T = np.transpose(eeg_data, (2, 1, 0))

    SEGMENT_LEN = 640

    def bandpass_filter(segment):
        nyq = 0.5 * FS
        b, a = sig.butter(4, [0.5/nyq, 50.0/nyq], btype='band')
        filtered = np.zeros_like(segment)
        for ch in range(segment.shape[1]):
            filtered[:, ch] = sig.filtfilt(b, a, segment[:, ch])
        return filtered

    def normalize(segment):
        normalized = np.zeros_like(segment)
        for ch in range(segment.shape[1]):
            d = segment[:, ch]
            normalized[:, ch] = (d - np.mean(d)) / (np.std(d) + 1e-8)
        return normalized

    print("Segmenting...")
    segments, seg_labels = [], []
    for i in range(data_T.shape[0]):
        subject = data_T[i]
        n_seg = subject.shape[0] // SEGMENT_LEN
        for j in range(n_seg):
            seg = subject[j*SEGMENT_LEN:(j+1)*SEGMENT_LEN, :]
            seg = bandpass_filter(seg)
            seg = normalize(seg)
            segments.append(seg)
            seg_labels.append(class_labels[i])

    X_raw = np.array(segments)
    y     = np.array(seg_labels)

    print(f"Extracting entropy features... ({len(X_raw)} segments)")
    print("This will take 10-15 minutes...")

    X_features = []
    for i, seg in enumerate(X_raw):
        feat = extract_features_v3(seg)
        X_features.append(feat)
        if (i+1) % 100 == 0:
            print(f"  {i+1}/{len(X_raw)} done...")

    X_features = np.array(X_features)
    print(f"Feature matrix shape: {X_features.shape}")

    np.save('data/X_features_v3.npy', X_features)
    np.save('data/y_labels.npy', y)
    print("Saved! X_features_v3.npy")


########## src\common.py ##########

"""Shared helpers. Run every script from anywhere: paths are relative to the project root."""
import os
from pathlib import Path
import numpy as np

ROOT = Path(__file__).resolve().parent.parent
DATA = ROOT / "data"
OUT = ROOT / "results"
OUT.mkdir(exist_ok=True)

N_SUB, SEG, SEED = 45, 30, 42
N_EST = int(os.environ.get("N_EST", 200))          # set N_EST=60 for a quick test run
FEATURE_FILE = "X_features_v3_fixed.npy"                 # final 252-feature matrix used by train_xgb.py
CH = ["AF3", "F7", "F3", "FC5", "T7", "P7", "O1", "O2", "P8", "T8", "FC6", "F4", "F8", "AF4"]


def load_xy(name=FEATURE_FILE):
    X = np.load(DATA / name)
    _, y = np.unique(np.load(DATA / "y_labels.npy"), return_inverse=True)
    groups = np.repeat(np.arange(N_SUB), SEG)        # assumes segments are ordered subject by subject
    assert X.shape[0] == len(y) == N_SUB * SEG, f"unexpected shapes: X{X.shape} y{y.shape}"
    for g in range(N_SUB):
        assert len(np.unique(y[groups == g])) == 1, (
            f"subject {g} has mixed labels -> segment order is not subject-major; fix 'groups'")
    return X, y, groups


def eta_sq_all(X, groups):
    """Share of each feature's variance explained by subject identity (1 = pure subject fingerprint)."""
    gm = X.mean(0)
    means = np.stack([X[groups == g].mean(0) for g in range(N_SUB)])
    ssb = SEG * ((means - gm) ** 2).sum(0)
    sst = ((X - gm) ** 2).sum(0)
    return np.where(sst > 1e-12, ssb / np.maximum(sst, 1e-12), np.nan)

import warnings; warnings.filterwarnings("ignore")
import numpy as np, pandas as pd, mne
from scipy.signal import butter, sosfiltfilt

fif = r'C:\Users\HP\Downloads\CollectedData_2021-07-29_1.fif'
times = pd.read_csv('snippet_eeg_times.csv')
STEW = ['AF3','F7','F3','FC5','T7','P7','O1','O2','P8','T8','FC6','F4','F8','AF4']
FS, WIN = 128, 640

raw = mne.io.read_raw_fif(fif, preload=False, verbose=False)
raw.pick(STEW).load_data()
raw.resample(FS)
data = raw.get_data().T                     # (time, 14)
sos = butter(4, [0.5, 50], btype="band", fs=FS, output="sos")
data = sosfiltfilt(sos, data, axis=0)       # STEW-এর মতোই 0.5-50 Hz

wins, owner = [], []
for _, r in times.iterrows():
    a, b = int(r.read_start_s * FS), int(r.read_end_s * FS)
    seg = data[a:b]
    for k in range(len(seg) // WIN):        # 5 সেকেন্ডের অ-ওভারল্যাপ উইন্ডো
        w = seg[k*WIN:(k+1)*WIN]
        wins.append((w - w.mean(0)) / (w.std(0) + 1e-8))   # প্রতি উইন্ডো, প্রতি চ্যানেল z-score
        owner.append(r.snippet)

np.savez('windows_p01.npz', X=np.stack(wins), snippet=np.array(owner))
print('windows:', np.stack(wins).shape)
print('std check (should be ~1):', np.stack(wins).std())
print('windows per snippet: min', min(np.unique(owner, return_counts=True)[1]),
      'max', max(np.unique(owner, return_counts=True)[1]))
